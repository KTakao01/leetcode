class Solution:
    def twoSum(self, nums: List[int], target: int) -> List[int]:
        # 総当たりを避け時間計算量を節約するためにハッシュマップに配列とインデックスを格納する
        nums_to_indices = {}
        # ハッシュマップに一回ループを回してデータを保存する
        for i in range(len(nums)):
            nums_to_indices[nums[i]] = i
        # ハッシュマップに保存したデータを活用する。
        for i in range(len(nums)):
            # 目標の数はそのまま配列に含まれておらず、扱いにくい。補数=目標の数-ハッシュマップのインデックスの値とすることで、どちらもリストに関する値となる。
            complement = target - nums[i]
            # ハッシュマップの中に補数が含まれていて、かつ、補数のインデックスを重複して使っていない
            if complement in nums_to_indices and i != nums_to_indices[complement] :
                return [i,nums_to_indices[complement]]
